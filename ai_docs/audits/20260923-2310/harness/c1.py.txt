def P(path: str, needle: str, delta: int = 0, nth: int = 1) -> dict[str, int]:
    lines = Path(path).read_text(encoding="utf-8").splitlines()
    seen = 0
    for i, ln in enumerate(lines):
        c = ln.find(needle)
        while c != -1:
            seen += 1
            if seen == nth:
                return {"line": i, "character": c + delta}
            c = ln.find(needle, c + 1)
    raise KeyError(needle)


def span(path: str, needle: str) -> dict[str, int]:
    p = P(path, needle)
    return {"start_line": p["line"], "start_character": p["character"], "end_line": p["line"],
            "end_character": p["character"] + len(needle)}


def fa(path: str, needle: str, delta: int = 0, **kw: Any) -> dict[str, Any]:
    return {"file_path": path, **P(path, needle, delta), **kw}

# ── variant generation ─────────────────────────────────────────────────────
INT_EDGE_KEYS = ("line", "character", "limit", "offset", "depth", "index", "min_lines", "max_items", "count",
                 "start_line", "end_line")
STR_EDGE = {"empty": "", "unicode": "ünï_名前_🐍", "long": "x" * 5000, "ws": "   ", "punct": "not an identifier!!"}
PATH_EDGE = {"empty": "", "missing": str(WS / "src" / "does_not_exist.py"), "relative": "src/audit.py",
             "unicode_missing": str(WS / "src" / "ünï_文件.py"), "dir": str(WS / "src"),
             "outside_ws": "C:/Windows/win.ini", "long": "C:/" + "a" * 5000 + ".py"}


def base_type(prop: dict) -> str:
    if "anyOf" in prop:
        ts = [x.get("type") for x in prop["anyOf"] if x.get("type") != "null"]
        return ts[0] if ts else "?"
    return prop.get("type", "?")


def variants(name: str, schema: dict, valid: dict | None) -> list[tuple[str, dict]]:
    out: list[tuple[str, dict]] = []
    props = schema.get("properties", {})
    req = [r for r in schema.get("required", [])]
    base = valid if valid is not None else {}
    if req:
        out.append(("missing_all_required", {}))
        if valid is not None and len(req) > 1:
            v = dict(base); v.pop(req[0], None); out.append((f"missing:{req[0]}", v))
        v = dict(base); v[req[0]] = None; out.append((f"null:{req[0]}", v))
    if valid is None:
        return out
    # wrong-type: first int, first str, first array, first bool (never `apply`)
    done = set()
    for k, p in props.items():
        t = base_type(p)
        if k == "apply" or t in done:
            continue
        bad = {"integer": "abc", "string": 12345, "array": "not-a-list", "boolean": "maybe", "object": [1, 2]}.get(t)
        if bad is None:
            continue
        v = dict(base); v[k] = bad; out.append((f"wrongtype:{k}", v)); done.add(t)
    v = dict(base); v["__unknown_extra__"] = 1; out.append(("extra_unknown_arg", v))
    for k in INT_EDGE_KEYS:
        if k in props and base_type(props[k]) == "integer":
            v = dict(base); v[k] = -1; out.append((f"neg:{k}", v))
            if k in ("line", "start_line", "character", "limit", "depth", "count", "offset"):
                v = dict(base); v[k] = 10**9; out.append((f"huge:{k}", v))
    str_keys = [k for k, p in props.items() if base_type(p) == "string" and k in base]
    for k in str_keys[:2]:
        table = PATH_EDGE if ("path" in k or "file" in k or k in ("source_file", "destination_package", "root_path", "output_dir")) else STR_EDGE
        for tag, val in table.items():
            v = dict(base); v[k] = val; out.append((f"str:{k}:{tag}", v))
    arr_keys = [k for k, p in props.items() if base_type(p) == "array" and k in base]
    for k in arr_keys[:1]:
        v = dict(base); v[k] = []; out.append((f"emptylist:{k}", v))
        nested: Any = ["x"]
        for _ in range(60):
            nested = [nested]
        v = dict(base); v[k] = nested; out.append((f"deepnest:{k}", v))
    return out


# ── workspace side-effect detection ────────────────────────────────────────
def ws_hash(root: Path = WS) -> dict[str, str]:
    h = {}
    for f in root.rglob("*"):
        if f.is_file() and ".ropeproject" not in f.parts and "__pycache__" not in f.parts:
            h[str(f.relative_to(root))] = hashlib.sha1(f.read_bytes()).hexdigest()
    return h


def restore() -> None:
    for f in list(WS.iterdir()):
        if f.name == ".ropeproject":
            continue
        shutil.rmtree(f) if f.is_dir() else f.unlink()
    for f in PRISTINE.iterdir():
        if f.name == ".ropeproject":
            continue
        (shutil.copytree(f, WS / f.name) if f.is_dir() else shutil.copy2(f, WS / f.name))


def summarize(res: Any) -> dict[str, Any]:
    text = " ".join(getattr(c, "text", "") for c in (res.content or []))
    sc = res.structured_content
    return {"is_error": bool(res.is_error), "text": text[:600], "text_len": len(text),
            "has_traceback": "Traceback (most recent call last)" in text,
            "structured": sc}


class Runner:
    def __init__(self) -> None:
        self.records: list[dict[str, Any]] = []
        self.pristine_hash = ws_hash()

    async def call(self, s: Any, profile: str, tool: str, variant: str, args: dict) -> dict[str, Any]:
        rec: dict[str, Any] = {"profile": profile, "tool": tool, "variant": variant,
                               "args": _trim(args)}
        before = ws_hash()
        t0 = time.perf_counter()
        try:
            res = await asyncio.wait_for(s.call_tool(tool, args, read_timeout_seconds=TIMEOUT), TIMEOUT + 5)
            rec.update(outcome="error_result" if res.is_error else "ok", **summarize(res))
        except TimeoutError:
            rec.update(outcome="timeout")
        except RuntimeError as exc:
            msg = str(exc)
            rec.update(outcome="output_schema_violation" if "structured content" in msg or "output schema" in msg
                       else "runtime_error", text=msg[:800])
        except Exception as exc:  # JSON-RPC error or transport death
            rec.update(outcome="protocol_error", exc_type=type(exc).__name__, text=str(exc)[:800],
                       has_traceback="Traceback" in str(exc))
        rec["seconds"] = round(time.perf_counter() - t0, 3)
        after = ws_hash()
        if after != before:
            changed = sorted(k for k in set(before) | set(after) if before.get(k) != after.get(k))
            rec["ws_changed"] = changed
            restore()
        if rec.get("structured") is not None:
            blob = json.dumps(rec["structured"], default=str)
            rec["structured_len"] = len(blob)
            rec["structured"] = json.loads(blob) if len(blob) < 4000 else blob[:4000]
        self.records.append(rec)
        flag = rec["outcome"] + (" WS_CHANGED" if "ws_changed" in rec else "")
        print(f"{rec['seconds']:7.2f}s {profile[:3]} {tool:28s} {variant:34s} {flag}", flush=True)
        return rec


def _trim(obj: Any) -> Any:
    s = json.dumps(obj, default=str)
    return obj if len(s) < 600 else s[:600] + "…"


async def run_profile(profile: str, names: list[str], schemas: dict, valid: dict, runner: Runner) -> None:
    async with session(profile) as (s, _):
        await s.list_tools()
        for name in names:
            v = valid.get(name)
            if v is not None:
                r1 = await runner.call(s, profile, name, "valid", v)
                if schemas[name].get("annotations", {}).get("readOnlyHint") and r1["outcome"] == "ok":
                    r2 = await runner.call(s, profile, name, "valid_repeat", v)
                    r1["idempotent"] = r1.get("structured") == r2.get("structured")
            for tag, args in variants(name, schemas[name]["inputSchema"], v):
                await runner.call(s, profile, name, tag, args)


async def run_sequence(runner: Runner) -> None:

    async with session("refactoring") as (s, _):

        await c(s, "refactoring", "undo_refactoring", "seq:undo_count_neg", {"count": -1})

        # profile gating: analysis-only tool must be unknown on refactoring profile
        await c(s, "refactoring", "call_hierarchy", "seq:profile_gated", fa(A, "helper(a"))
        await c(s, "refactoring", "no_such_tool", "seq:unknown_tool", {})


async def main() -> None:
    schemas: dict[str, dict] = {}
    prof_of: dict[str, list[str]] = {}
    for prof in ("refactoring", "analysis"):
        for t in json.loads((RAW / f"inventory_{prof}.json").read_text(encoding="utf-8"))["tools"]:
            schemas.setdefault(t["name"], t)
            prof_of.setdefault(t["name"], []).append(prof)
    valid = valid_args()
    missing = set(schemas) - set(valid)
    assert not missing, missing
    only = set(sys.argv[1:])
    runner = Runner()
    seq_tools = {k for k, v in valid.items() if v is None}
    ref = sorted(n for n in schemas if "refactoring" in prof_of[n] and n not in seq_tools and (not only or n in only))
    ana = sorted(n for n in schemas if prof_of[n] == ["analysis"] and (not only or n in only))
    await run_profile("analysis", ana, schemas, valid, runner)
    await run_profile("refactoring", ref, schemas, valid, runner)
    if not only:
        await run_sequence(runner)
    final = ws_hash()
    out = RAW / ("c1_calls.json" if not only else "c1_calls_partial.json")
    out.write_text(json.dumps({"records": runner.records, "ws_final_matches_pristine": final == runner.pristine_hash},
                              indent=1, default=str), encoding="utf-8")
    print("records", len(runner.records), "final ws pristine:", final == runner.pristine_hash)


asyncio.run(main())
