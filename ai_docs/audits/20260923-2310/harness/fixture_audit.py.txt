class Shape:
    """Base shape."""

    def __init__(self, width: float, height: float) -> None:
        self.width = width
        self.height = height

    def area(self) -> float:
        scale = 1.0
        result = self.width * self.height * scale
        return result

    def describe(self, prefix: str = "shape") -> str:
        return f"{prefix}: {self.area()}"


class Square(Shape):
    def area(self) -> float:
        return self.width * self.width


class HasArea(Protocol):
    def area(self) -> float: ...

def helper(a: int, b: int = 2) -> int:
    """Add two numbers.

    Args:
        a: first.
